Indexes:
    "biz_receivable_pkey" PRIMARY KEY, btree (id)
    "idx_biz_ar_customer" btree (customer_id, status)
    "idx_biz_ar_due" btree (due_date, status)
    "idx_biz_ar_period" btree (period_code)
    "idx_biz_ar_status" btree (status)
    "idx_biz_ar_voucher" btree (voucher_id) WHERE voucher_id IS NOT NULL
    "uk_biz_ar_no" UNIQUE, btree (ar_no) WHERE deleted_at IS NULL

Check constraints:
    "ck_biz_ar_amount" CHECK (amount > 0::numeric AND received_amount >= 0::numeric AND received_amount <= amount)
    "ck_biz_ar_credit" CHECK (credit_days >= 0)
    "ck_biz_ar_settled" CHECK (status::text <> 'SETTLED'::text OR unsettled_amount = 0::numeric)
    "ck_biz_ar_status" CHECK (status::text = ANY (ARRAY['OPEN'::character varying, 'PARTIAL'::character varying, 'SETTLED'::character varying, 'WRITTEN_OFF'::character varying]::text[]))
    "ck_biz_ar_unsettled" CHECK (unsettled_amount = (amount - received_amount))
