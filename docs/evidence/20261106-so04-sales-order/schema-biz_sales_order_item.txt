                         Table "public.biz_sales_order_item"
    Column     |           Type           | Collation | Nullable |      Default      
---------------+--------------------------+-----------+----------+-------------------
 id            | uuid                     |           | not null | gen_random_uuid()
 so_id         | uuid                     |           | not null | 
 line_no       | integer                  |           | not null | 
 material_id   | uuid                     |           | not null | 
 qty           | numeric(18,4)            |           | not null | 
 price         | numeric(18,6)            |           | not null | 0
 amount        | numeric(18,4)            |           | not null | 0
 tax_rate      | numeric(7,4)             |           | not null | 0
 tax_amount    | numeric(18,4)            |           | not null | 0
 delivered_qty | numeric(18,4)            |           | not null | 0
 created_at    | timestamp with time zone |           | not null | CURRENT_TIMESTAMP
 updated_at    | timestamp with time zone |           | not null | CURRENT_TIMESTAMP
Indexes:
    "biz_sales_order_item_pkey" PRIMARY KEY, btree (id)
    "idx_biz_sales_order_item_material" btree (material_id)
    "uk_biz_sales_order_item_line" UNIQUE, btree (so_id, line_no)
Check constraints:
    "ck_sales_order_item_amount" CHECK (amount >= 0::numeric AND tax_amount >= 0::numeric)
    "ck_sales_order_item_delivered" CHECK (delivered_qty >= 0::numeric AND delivered_qty <= qty)
    "ck_sales_order_item_line" CHECK (line_no >= 1)
    "ck_sales_order_item_price" CHECK (price >= 0::numeric)
    "ck_sales_order_item_qty" CHECK (qty > 0::numeric)
    "ck_sales_order_item_rate" CHECK (tax_rate >= 0::numeric AND tax_rate <= 100::numeric)
Foreign-key constraints:
    "fk_biz_sales_order_item_order" FOREIGN KEY (so_id) REFERENCES biz_sales_order(id) ON DELETE CASCADE
