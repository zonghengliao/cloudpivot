/// 行级删除判据：上传人本人 ∪ 持 `business:attachment:delete`。
///
/// # 参数
///
/// - `claims`：当前用户；
/// - `uploaded_by`：附件上传人（历史行可能为 `NULL` —— 无属主行按「只有升权码能删」处理）。
pub fn can_delete(claims: &TokenClaims, uploaded_by: Option<Uuid>) -> bool {
    if claims.has_permission(PERM_DELETE) {
        return true;
    }
    match (uploaded_by, crate::utils::current_user_uuid(claims)) {
        (Some(owner), Some(me)) => owner == me,
        _ => false,
    }
}

/// # 校验顺序（每一步都先于「更贵的那一步」）
///
/// 1. `Content-Length` 与服务端硬顶（**读流之前**，否则大文件会先读进内存再被拒）；

/// 3. 文本字段白名单（`doc_type` / `doc_id` / `file_role` / `remark`）；
/// 4. **单据读码**（403，[`dto::can_view_doc`]，ATT-04）—— 与 `list` / `download` 同一把尺子：
///    看不见这张单，就别往上挂东西（否则会往别人的单据里投毒，并能用 404/400 差异探测
///    单据是否存在）。第 3、4 步合在 [`dto::preflight_upload`] 里，**不查库**；
/// 5. 单据存在性（404 —— 不允许给不存在的单据挂号）：排在 403 **之后**，
