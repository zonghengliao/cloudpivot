 SELECT it.id AS agreement_item_id,
    it.agreement_id,
    a.agreement_no,
    it.line_no,
    it.material_id,
    COALESCE(m.material_code, ''::character varying) AS material_code,
    COALESCE(m.material_name, ''::character varying) AS material_name,
    it.unit_id,
    COALESCE(u.unit_name, ''::character varying) AS unit_name,
    it.agreed_price,
    it.tax_rate,
    a.price_includes_tax,
    a.price_currency,
    it.max_qty,
    COALESCE(usg.qty, 0::numeric) AS executed_qty,
    it.executed_qty AS booked_qty,
    COALESCE(usg.amount, 0::numeric) AS executed_amount,
    COALESCE(usg.with_tax, 0::numeric) AS executed_amount_with_tax,
        CASE
            WHEN a.price_includes_tax THEN COALESCE(usg.with_tax, 0::numeric)
            ELSE COALESCE(usg.amount, 0::numeric)
        END AS executed_amount_capped,
    it.max_qty - COALESCE(usg.qty, 0::numeric) AS remaining_qty,
    it.max_qty * it.agreed_price AS cap_amount,
    it.max_qty * it.agreed_price -
        CASE
            WHEN a.price_includes_tax THEN COALESCE(usg.with_tax, 0::numeric)
            ELSE COALESCE(usg.amount, 0::numeric)
        END AS remaining_amount,
    round(COALESCE(usg.qty, 0::numeric) / it.max_qty, 6) AS exec_qty_rate,
    round(
        CASE
            WHEN a.price_includes_tax THEN COALESCE(usg.with_tax, 0::numeric)
            ELSE COALESCE(usg.amount, 0::numeric)
        END / NULLIF(it.max_qty * it.agreed_price, 0::numeric), 6) AS exec_amount_rate,
    COALESCE(usg.doc_count, 0::bigint) AS order_doc_count,
    COALESCE(usg.line_count, 0::bigint) AS order_line_count,
    usg.last_order_on,
        CASE
            WHEN it.max_qty IS NULL THEN 'UNLIMITED'::text
            WHEN COALESCE(usg.qty, 0::numeric) > it.max_qty THEN 'OVER'::text
            WHEN COALESCE(usg.qty, 0::numeric) >= it.max_qty THEN 'DONE'::text
            WHEN COALESCE(usg.qty, 0::numeric) > 0::numeric THEN 'PARTIAL'::text
            ELSE 'NOT_STARTED'::text
        END AS exec_state
   FROM biz_framework_agreement_item it
     JOIN biz_framework_agreement a ON a.id = it.agreement_id AND a.deleted_at IS NULL
     LEFT JOIN biz_material m ON m.id = it.material_id AND m.deleted_at IS NULL
     LEFT JOIN biz_unit u ON u.id = it.unit_id
     LEFT JOIN ( SELECT i.framework_agreement_item_id AS item_id,
            sum(i.qty) AS qty,
            sum(i.amount) AS amount,
            sum(i.with_tax_amount) AS with_tax,
            count(DISTINCT o.id) AS doc_count,
            count(*) AS line_count,
            max(o.order_date) AS last_order_on
           FROM biz_purchase_order_item i
             JOIN biz_purchase_order o ON o.id = i.order_id AND o.deleted_at IS NULL AND o.status::text <> 'CANCELLED'::text
          WHERE i.framework_agreement_item_id IS NOT NULL
          GROUP BY i.framework_agreement_item_id) usg ON usg.item_id = it.id;

 WITH item_agg AS (
         SELECT e.agreement_id,
            count(*) AS item_count,
            count(*) FILTER (WHERE e.exec_state = 'UNLIMITED'::text) AS unlimited_items,
            count(*) FILTER (WHERE e.exec_state = 'OVER'::text) AS over_items,
            count(*) FILTER (WHERE e.exec_state = 'DONE'::text) AS done_items,
            count(*) FILTER (WHERE e.exec_state = 'PARTIAL'::text) AS partial_items,
            count(*) FILTER (WHERE e.exec_state = 'NOT_STARTED'::text) AS not_started_items,
            COALESCE(sum(e.executed_qty), 0::numeric) AS executed_qty,
            COALESCE(sum(e.executed_amount), 0::numeric) AS executed_amount,
            COALESCE(sum(e.executed_amount_with_tax), 0::numeric) AS executed_amount_with_tax,
            COALESCE(sum(e.executed_amount_capped), 0::numeric) AS executed_amount_capped,
            COALESCE(sum(e.booked_qty), 0::numeric) AS booked_qty,
            sum(e.max_qty) FILTER (WHERE e.max_qty IS NOT NULL) AS item_max_qty_sum_raw
           FROM v_framework_agreement_item_execution e
          GROUP BY e.agreement_id
        ), po_agg AS (
         SELECT o.framework_agreement_id AS agreement_id,
            count(DISTINCT o.id) FILTER (WHERE o.deleted_at IS NULL AND o.status::text <> 'CANCELLED'::text) AS order_doc_count,
            count(DISTINCT o.id) FILTER (WHERE o.deleted_at IS NOT NULL OR o.status::text = 'CANCELLED'::text) AS released_doc_count,
            max(o.order_date) FILTER (WHERE o.deleted_at IS NULL AND o.status::text <> 'CANCELLED'::text) AS last_order_on
           FROM biz_purchase_order o
          WHERE o.framework_agreement_id IS NOT NULL
          GROUP BY o.framework_agreement_id
        ), base AS (
         SELECT a.id AS agreement_id,
            a.agreement_no,
            a.agreement_name,
            a.supplier_id,
            a.supplier_name,
            COALESCE(s.supplier_name, a.supplier_name, ''::character varying) AS supplier_display_name,
            a.dept_id,
            a.dept_name,
            a.owner_id,
            a.owner_name,
            a.start_date,
            a.end_date,
            a.sign_date,
            a.status,
            a.price_mode,
            a.price_includes_tax,
            a.price_currency,
            a.price_tolerance,
            a.total_max_qty,
            a.total_max_amount,
            a.total_executed_qty AS booked_total_qty,
            a.total_executed_amount AS booked_total_amount,
            COALESCE(i.item_count, 0::bigint) AS item_count,
            COALESCE(i.unlimited_items, 0::bigint) AS unlimited_items,
            COALESCE(i.over_items, 0::bigint) AS over_items,
            COALESCE(i.done_items, 0::bigint) AS done_items,
            COALESCE(i.partial_items, 0::bigint) AS partial_items,
            COALESCE(i.not_started_items, 0::bigint) AS not_started_items,
            COALESCE(i.executed_qty, 0::numeric) AS executed_qty,
            COALESCE(i.executed_amount, 0::numeric) AS executed_amount,
            COALESCE(i.executed_amount_with_tax, 0::numeric) AS executed_amount_with_tax,
            COALESCE(i.executed_amount_capped, 0::numeric) AS executed_amount_capped,
            COALESCE(i.booked_qty, 0::numeric) AS booked_qty,
                CASE
                    WHEN COALESCE(i.unlimited_items, 0::bigint) = 0 THEN COALESCE(i.item_max_qty_sum_raw, 0::numeric)
                    ELSE NULL::numeric
                END AS item_max_qty_sum,
            COALESCE(p.order_doc_count, 0::bigint) AS order_doc_count,
            COALESCE(p.released_doc_count, 0::bigint) AS released_doc_count,
            p.last_order_on,
            a.created_at,
            a.end_date - CURRENT_DATE AS days_left,
            a.status::text = 'APPROVED'::text AND CURRENT_DATE >= a.start_date AND CURRENT_DATE <= a.end_date AS is_effective,
                CASE
                    WHEN a.status::text = ANY (ARRAY['DRAFT'::character varying, 'SUBMITTED'::character varying]::text[]) THEN a.status
                    WHEN CURRENT_DATE < a.start_date THEN 'NOT_STARTED'::character varying
                    WHEN CURRENT_DATE > a.end_date THEN 'EXPIRED'::character varying
                    ELSE 'ACTIVE'::character varying
                END AS validity_state,
            COALESCE(a.total_max_qty,
                CASE
                    WHEN COALESCE(i.unlimited_items, 0::bigint) = 0 THEN COALESCE(i.item_max_qty_sum_raw, 0::numeric)
                    ELSE NULL::numeric
                END) AS cap_qty,
            a.total_max_amount AS cap_amount
           FROM biz_framework_agreement a
             LEFT JOIN biz_supplier s ON s.id = a.supplier_id AND s.deleted_at IS NULL
             LEFT JOIN item_agg i ON i.agreement_id = a.id
             LEFT JOIN po_agg p ON p.agreement_id = a.id
          WHERE a.deleted_at IS NULL
        )

    executed_amount,

    executed_amount_capped,
    booked_qty,
    booked_total_qty,
    booked_total_amount,
    cap_qty - executed_qty AS remaining_qty,
    cap_amount - executed_amount_capped AS remaining_amount,
    round(executed_qty / cap_qty, 6) AS exec_qty_rate,
    round(executed_amount_capped / cap_amount, 6) AS exec_amount_rate,
    order_doc_count,
    released_doc_count,
    last_order_on,
    days_left,
    is_effective,
    validity_state,
        CASE
            WHEN cap_qty IS NULL AND cap_amount IS NULL THEN 'UNLIMITED'::text
            WHEN cap_qty IS NOT NULL THEN
            CASE
                WHEN executed_qty > cap_qty THEN 'OVER'::text
                WHEN executed_qty >= cap_qty THEN 'DONE'::text
                WHEN executed_qty > 0::numeric THEN 'PARTIAL'::text
                ELSE 'NOT_STARTED'::text
            END
            ELSE
            CASE
                WHEN executed_amount_capped > cap_amount THEN 'OVER'::text
                WHEN executed_amount_capped >= cap_amount THEN 'DONE'::text
                WHEN executed_amount_capped > 0::numeric THEN 'PARTIAL'::text
                ELSE 'NOT_STARTED'::text
            END
        END AS exec_state,
    created_at
   FROM base b;
