 currency_matched | boolean                  |           |          |         | plain    | 单据币种是否等于合同币种：false = 币种串了（守卫也不看币种，报表只显形不换算、不剔除）
View definition:
 SELECT o.contract_id,
    'PURCHASE_ORDER'::text AS source_type,
    o.id AS order_id,
    o.order_no,
    o.status AS order_status,
    o.order_date,
    o.currency_code::text AS currency_code,
    o.total_amount,
    o.total_with_tax,
    o.deleted_at AS order_deleted_at,
    o.deleted_at IS NULL AND o.status::text <> 'CANCELLED'::text AS counted,
    o.currency_code::text = c.currency_code::text AS currency_matched
   FROM biz_purchase_order o
     JOIN biz_contract c ON c.id = o.contract_id
  WHERE o.contract_id IS NOT NULL
UNION ALL
 SELECT s.contract_id,
    'SALES_ORDER'::text AS source_type,
    s.id AS order_id,
    s.so_no AS order_no,
    s.status AS order_status,
    s.order_date,
    s.currency_code::text AS currency_code,
    s.total_amount,
    s.total_with_tax,
    s.deleted_at AS order_deleted_at,
    s.deleted_at IS NULL AND s.status::text <> 'CANCELLED'::text AS counted,
    s.currency_code::text = c.currency_code::text AS currency_matched
   FROM biz_sales_order s
     JOIN biz_contract c ON c.id = s.contract_id
  WHERE s.contract_id IS NOT NULL;


=== 2) v_contract_execution 定义 ===
                                                                                                                 View "public.v_contract_execution"

-----------------------------+--------------------------+-----------+----------+---------+----------+----------------------------------------------------------------------------------------------------------------------------------------------------------------

View definition:
 WITH agg AS (
         SELECT e.contract_id,
            count(*) FILTER (WHERE e.counted) AS order_doc_count,
            count(*) FILTER (WHERE NOT e.counted) AS released_doc_count,
            count(*) FILTER (WHERE e.counted AND NOT e.currency_matched) AS currency_mismatch_doc_count,
            count(*) FILTER (WHERE e.counted AND e.source_type = 'PURCHASE_ORDER'::text) AS purchase_doc_count,
            count(*) FILTER (WHERE e.counted AND e.source_type = 'SALES_ORDER'::text) AS sales_doc_count,
            COALESCE(sum(e.total_with_tax) FILTER (WHERE e.counted), 0::numeric) AS executed_amount,
            COALESCE(sum(e.total_amount) FILTER (WHERE e.counted), 0::numeric) AS executed_amount_net,
            max(e.order_date) FILTER (WHERE e.counted) AS last_order_on
           FROM v_contract_execution_order e
          GROUP BY e.contract_id
        ), base AS (
         SELECT c.id AS contract_id,
            c.contract_no,
            c.contract_name,
            c.contract_type,
            c.status,
            c.party_a_id,
            c.party_a_name,
            c.party_b_type,
            c.party_b_id,
            c.party_b_name,
            COALESCE(s.supplier_name, cu.name) AS party_b_current_name,
            c.owner_id,
            u.real_name AS owner_name,
            c.dept_id,
            d.dept_name,
            c.amount,
            c.currency_code::text AS currency_code,
            c.sign_date,
            c.start_date,
            c.end_date,
            c.created_at,
            c.updated_at,
                CASE c.contract_type
                    WHEN 'PURCHASE'::text THEN 'PURCHASE_ORDER'::text
                    WHEN 'SALES'::text THEN 'SALES_ORDER'::text
                    ELSE NULL::text
                END AS downstream_type,
            COALESCE(a.order_doc_count, 0::bigint) AS order_doc_count,
            COALESCE(a.released_doc_count, 0::bigint) AS released_doc_count,
            COALESCE(a.currency_mismatch_doc_count, 0::bigint) AS currency_mismatch_doc_count,
            COALESCE(a.purchase_doc_count, 0::bigint) AS purchase_doc_count,
            COALESCE(a.sales_doc_count, 0::bigint) AS sales_doc_count,
            COALESCE(a.executed_amount, 0::numeric) AS executed_amount,
            COALESCE(a.executed_amount_net, 0::numeric) AS executed_amount_net,
            a.last_order_on,
            c.end_date - CURRENT_DATE AS days_left,
            c.end_date IS NOT NULL AND c.end_date < CURRENT_DATE AS expired,
            c.status::text = ANY (ARRAY['SIGNED'::character varying, 'ACTIVE'::character varying]::text[]) AS usable_for_order
           FROM biz_contract c
             LEFT JOIN biz_supplier s ON s.id = c.party_b_id AND c.party_b_type::text = 'SUPPLIER'::text
             LEFT JOIN biz_customer cu ON cu.id = c.party_b_id AND c.party_b_type::text = 'CUSTOMER'::text
             LEFT JOIN sys_user u ON u.id = c.owner_id
             LEFT JOIN sys_dept d ON d.id = c.dept_id
             LEFT JOIN agg a ON a.contract_id = c.id
          WHERE c.deleted_at IS NULL
        )
 SELECT contract_id,
    contract_no,
    contract_name,
    contract_type,
    status,
    party_a_id,
    party_a_name,
    party_b_type,
    party_b_id,
    party_b_name,
    party_b_current_name,
    owner_id,
    owner_name,
    dept_id,
    dept_name,
    amount,
    currency_code,
    sign_date,
    start_date,
    end_date,
    downstream_type,
    order_doc_count,
    released_doc_count,
    currency_mismatch_doc_count,
    purchase_doc_count,
    sales_doc_count,
    executed_amount,
    executed_amount_net,
    amount - executed_amount AS remaining_amount,
    round(executed_amount / amount, 6) AS exec_amount_rate,
        CASE
            WHEN downstream_type IS NULL THEN 'NO_DOWNSTREAM'::text
            WHEN executed_amount > amount THEN 'OVER'::text
            WHEN executed_amount >= amount THEN 'DONE'::text
            WHEN executed_amount > 0::numeric THEN 'PARTIAL'::text
            ELSE 'NOT_STARTED'::text
        END AS exec_state,
    last_order_on,
    days_left,
    expired,
    usable_for_order,
    created_at,
    updated_at
   FROM base b;
